     public static String normalizeMessageType(String type) {
         if (type == null) return "NORMAL";
         switch (type.toUpperCase()) {
             case "ANNOUNCEMENT":
                 return "ANNOUNCEMENT";
             case "DIRECT":
                 return "DIRECT";
             case "TRAINER":
                 return "TRAINER";

                 return "NORMAL";
         }
     }
